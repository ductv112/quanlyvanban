/**
 * sync-to-excel.ts — Sync test results back to Excel template
 *
 * Input:
 *   - tests/results/playwright-results.json (Playwright JSON reporter)
 *   - tests/results/vitest-integration.json (Vitest JSON reporter, optional)
 *   - tests/results/vitest-unit.json (Vitest JSON reporter, optional)
 *   - docs/hdsd/20260505_Testcase_QLVB_V2.xlsx (template)
 *
 * Output:
 *   - docs/hdsd/<YYYYMMDD>_Testcase_QLVB_V2_results.xlsx (5 cột mới: Trạng thái, Run date, Duration, Error msg, Trace link)
 *
 * Mapping:
 *   - test title regex `^(TC-[A-Z0-9-]+)` extract TC-ID
 *   - TC-ID match column A của Excel template
 *   - Set 5 cột mới sau cột cuối cùng
 *
 * Coverage report (stdout):
 *   Total TC trong Excel: 847
 *   Auto coverage: 30 (3.5%)
 *   Not run: 817
 *
 * Usage:
 *   tsx tools/test-report/sync-to-excel.ts
 *   # Custom path:
 *   PLAYWRIGHT_RESULTS=./custom/pw.json EXCEL_TEMPLATE=./template.xlsx tsx sync-to-excel.ts
 */

import ExcelJS from 'exceljs';
import path from 'path';
import fs from 'fs';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);
const REPO_ROOT = path.resolve(__dirname, '..', '..');

const PLAYWRIGHT_RESULTS = process.env.PLAYWRIGHT_RESULTS || path.join(REPO_ROOT, 'tests', 'results', 'playwright-results.json');
const VITEST_INTEGRATION_RESULTS = process.env.VITEST_INTEGRATION_RESULTS || path.join(REPO_ROOT, 'e_office_app_new', 'backend', 'tests', 'results', 'vitest-integration.json');
const VITEST_UNIT_RESULTS = process.env.VITEST_UNIT_RESULTS || path.join(REPO_ROOT, 'e_office_app_new', 'backend', 'tests', 'results', 'vitest-unit.json');
const EXCEL_TEMPLATE = process.env.EXCEL_TEMPLATE || path.join(REPO_ROOT, 'docs', 'hdsd', '20260505_Testcase_QLVB_V2.xlsx');
const OUTPUT_DIR = process.env.OUTPUT_DIR || path.join(REPO_ROOT, 'docs', 'hdsd');

interface TestResult {
  tcId: string;
  status: 'Pass' | 'Fail' | 'Skip';
  duration: number; // ms
  error?: string;
  traceUrl?: string;
}

const TC_ID_REGEX = /^(TC-[A-Z0-9-]+)/;

function parsePlaywrightJson(filePath: string): TestResult[] {
  if (!fs.existsSync(filePath)) {
    console.warn(`[sync] Playwright results file missing: ${filePath} — skipping`);
    return [];
  }
  const raw = JSON.parse(fs.readFileSync(filePath, 'utf8'));
  const results: TestResult[] = [];

  function walkSuite(suite: { suites?: unknown[]; specs?: unknown[] }): void {
    (suite.suites || []).forEach((s) => walkSuite(s as { suites?: unknown[]; specs?: unknown[] }));
    (suite.specs || []).forEach((spec) => {
      const s = spec as { title: string; tests?: { results?: { status: string; duration: number; error?: { message?: string }; attachments?: { name: string; path?: string }[] }[] }[] };
      const m = s.title.match(TC_ID_REGEX);
      if (!m) return;
      const tcId = m[1];

      const test = s.tests?.[0];
      const result = test?.results?.[0];
      if (!result) {
        results.push({ tcId, status: 'Skip', duration: 0 });
        return;
      }
      const status = result.status === 'passed' ? 'Pass' : result.status === 'skipped' ? 'Skip' : 'Fail';
      const trace = result.attachments?.find((a) => a.name === 'trace');
      results.push({
        tcId,
        status,
        duration: result.duration || 0,
        error: status === 'Fail' ? (result.error?.message || '').slice(0, 200) : undefined,
        traceUrl: trace?.path,
      });
    });
  }

  if (raw.suites) {
    raw.suites.forEach((s: unknown) => walkSuite(s as { suites?: unknown[]; specs?: unknown[] }));
  }
  return results;
}

function parseVitestJson(filePath: string): TestResult[] {
  if (!fs.existsSync(filePath)) {
    console.warn(`[sync] Vitest results file missing: ${filePath} — skipping`);
    return [];
  }
  const raw = JSON.parse(fs.readFileSync(filePath, 'utf8'));
  const results: TestResult[] = [];

  // Vitest JSON shape: { testResults: [{ assertionResults: [{ title, status, duration, failureMessages }] }] }
  (raw.testResults || []).forEach((tr: { assertionResults?: { title?: string; fullName?: string; status?: string; duration?: number; failureMessages?: string[] }[] }) => {
    (tr.assertionResults || []).forEach((ar) => {
      const title = ar.fullName || ar.title || '';
      const m = title.match(TC_ID_REGEX);
      if (!m) return;
      const tcId = m[1];
      const status = ar.status === 'passed' ? 'Pass' : ar.status === 'skipped' ? 'Skip' : 'Fail';
      results.push({
        tcId,
        status,
        duration: ar.duration || 0,
        error: status === 'Fail' ? (ar.failureMessages?.[0] || '').slice(0, 200) : undefined,
      });
    });
  });
  return results;
}

async function syncToExcel(): Promise<void> {
  if (!fs.existsSync(EXCEL_TEMPLATE)) {
    throw new Error(`Excel template not found: ${EXCEL_TEMPLATE}`);
  }

  // Aggregate results
  const allResults = new Map<string, TestResult>();
  for (const r of parsePlaywrightJson(PLAYWRIGHT_RESULTS)) allResults.set(r.tcId, r);
  for (const r of parseVitestJson(VITEST_INTEGRATION_RESULTS)) {
    if (!allResults.has(r.tcId)) allResults.set(r.tcId, r);
  }
  for (const r of parseVitestJson(VITEST_UNIT_RESULTS)) {
    if (!allResults.has(r.tcId)) allResults.set(r.tcId, r);
  }

  console.log(`[sync] Parsed ${allResults.size} test results`);

  // Open Excel template
  const wb = new ExcelJS.Workbook();
  await wb.xlsx.readFile(EXCEL_TEMPLATE);
  const sheet = wb.worksheets[0];
  if (!sheet) throw new Error('No worksheet found in template');

  // Find header row + TC-ID column
  const headerRow = sheet.getRow(1);
  let tcIdCol = -1;
  headerRow.eachCell((cell, colNumber) => {
    const v = String(cell.value || '').toLowerCase();
    if (v.includes('tc-id') || v === 'tc id' || v === 'mã tc' || v.includes('test case id') || colNumber === 1) {
      if (tcIdCol === -1) tcIdCol = colNumber;
    }
  });
  if (tcIdCol === -1) tcIdCol = 1;
  console.log(`[sync] TC-ID column = ${tcIdCol}`);

  // Add 5 new columns after last column
  const lastCol = sheet.actualColumnCount;
  const STATUS_COL = lastCol + 1;
  const RUN_DATE_COL = lastCol + 2;
  const DURATION_COL = lastCol + 3;
  const ERROR_COL = lastCol + 4;
  const TRACE_COL = lastCol + 5;

  headerRow.getCell(STATUS_COL).value = 'Trạng thái';
  headerRow.getCell(RUN_DATE_COL).value = 'Run date';
  headerRow.getCell(DURATION_COL).value = 'Duration';
  headerRow.getCell(ERROR_COL).value = 'Error msg';
  headerRow.getCell(TRACE_COL).value = 'Trace link';

  // Iterate rows (start row 2)
  let totalTC = 0;
  let mappedCount = 0;
  let passCount = 0;
  let failCount = 0;
  let skipCount = 0;
  let notRunCount = 0;

  for (let rowIdx = 2; rowIdx <= sheet.actualRowCount; rowIdx++) {
    const row = sheet.getRow(rowIdx);
    const tcIdCell = row.getCell(tcIdCol);
    const tcIdValue = String(tcIdCell.value || '').trim();
    if (!tcIdValue) continue;
    totalTC++;

    const result = allResults.get(tcIdValue);
    const today = new Date().toISOString().slice(0, 10);

    if (!result) {
      row.getCell(STATUS_COL).value = 'Not run';
      notRunCount++;
      continue;
    }

    mappedCount++;
    row.getCell(STATUS_COL).value = result.status;
    row.getCell(RUN_DATE_COL).value = today;
    row.getCell(DURATION_COL).value = `${result.duration}ms`;

    if (result.status === 'Pass') passCount++;
    else if (result.status === 'Skip') skipCount++;
    else {
      failCount++;
      row.getCell(ERROR_COL).value = result.error || '';
      row.getCell(TRACE_COL).value = result.traceUrl || '';
    }
  }

  // Coverage report
  console.log('');
  console.log('='.repeat(60));
  console.log(`Total TC trong Excel: ${totalTC}`);
  console.log(`Mapped (auto coverage): ${mappedCount} (${((mappedCount / totalTC) * 100).toFixed(1)}%)`);
  console.log(`  Pass: ${passCount}`);
  console.log(`  Fail: ${failCount}`);
  console.log(`  Skip: ${skipCount}`);
  console.log(`Not run: ${notRunCount}`);
  console.log('='.repeat(60));

  // TC trong test nhưng không có trong Excel
  const excelTcIds = new Set<string>();
  for (let i = 2; i <= sheet.actualRowCount; i++) {
    const v = String(sheet.getRow(i).getCell(tcIdCol).value || '').trim();
    if (v) excelTcIds.add(v);
  }
  for (const [tcId] of allResults) {
    if (!excelTcIds.has(tcId)) {
      console.warn(`[sync] WARN: TC ${tcId} có trong test nhưng KHÔNG có trong Excel template`);
    }
  }

  // Write output
  const date = new Date().toISOString().slice(0, 10).replace(/-/g, '');
  const outputPath = path.join(OUTPUT_DIR, `${date}_Testcase_QLVB_V2_results.xlsx`);
  await wb.xlsx.writeFile(outputPath);
  console.log(`[sync] Written: ${outputPath}`);
}

syncToExcel().catch((e) => {
  console.error('[sync] FATAL:', e.message || e);
  process.exit(1);
});

## 7. Convention quan trọng

### Naming test

**Bắt buộc:** title bắt đầu bằng `TC-XXX-NNN ` để parser map về Excel.

```typescript
test('TC-AUTH-001 Đăng nhập thành công với tài khoản admin @smoke @P-High', ...);
```

### Tag

| Tag | Ý nghĩa | Filter |
|-----|---------|--------|
| `@smoke` | Smoke 30 TC P-High | `--grep @smoke` |
| `@regression` | Regression toàn bộ | `--grep @regression` |
| `@P-High` | Priority cao | `--grep @P-High` |
| `@flaky` | Quarantine flaky test | `--grep -v @flaky` |

### Storage state per role

```typescript
import { storageStateFor } from '../fixtures/auth';

test.use({ storageState: storageStateFor('vanthu') });
// Tự động có session vanthu — KHÔNG cần login
```

### Locator best practice

- ✅ `page.getByRole('button', { name: /Đăng nhập/i })`
- ✅ `page.locator('[data-testid="search-input"]')`
- ❌ `page.locator('text="Đăng nhập"')` (fragile khi UI thay đổi)
- ❌ `page.locator('.css-xyz123')` (CSS class hash thay đổi mỗi build)

## 8. Troubleshooting

### "Backend không phản hồi" trong globalSetup
→ Backend chưa chạy. Chạy `cd e_office_app_new/backend && PG_DATABASE=qlvb_test npm run dev` trước.

### "Login fail cho test_admin"
→ Test DB chưa seed fixture. Chạy `npm run test:db:reset`.

### "Cannot find module '@playwright/test'"
→ Chưa `npm install` ở root dir. Chạy `npm install` rồi `npx playwright install chromium`.

### Test pass local nhưng fail CI
→ CI dùng ubuntu-latest, có thể UI render khác. Check trace viewer trong CI artifact.

### File `qlvb_test` đã tồn tại nhưng test fail "fixture not found"
→ Schema cũ. Chạy `npm run test:db:reset` để DROP + CREATE + seed lại.
