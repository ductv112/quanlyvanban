 *      - database/seed/001_required_data.sql (admin + roles)
 *      - database/seed/003_test_fixtures.sql (5 user + 16 fixture)
 *   5. Verify: SELECT COUNT(*) FROM staff WHERE id IN (9001, 9002, 9003, 9004, 9005, 9099)
 *      → expect 6
 *
 * Usage:
 *   npm run test:db:reset
 *   # hoặc:
 *   tsx tools/test-db-reset.ts
 *
 * Env (đọc từ .env.test, fallback .env):

 *   PG_DATABASE (default: qlvb_test)

import { execSync } from 'child_process';
import { config as dotenvConfig } from 'dotenv';
import path from 'path';
import fs from 'fs';

// Load .env.test override .env
dotenvConfig({ path: path.resolve(process.cwd(), '.env.test'), override: true });
dotenvConfig({ path: path.resolve(process.cwd(), '.env'), override: false });

const PG_PORT = Number(process.env.PG_PORT) || 5432;
const PG_USER = process.env.PG_USER || 'qlvb_admin';
const PG_PASSWORD = process.env.PG_PASSWORD || 'QlvbDev@2026';
const PG_DATABASE = process.env.PG_DATABASE || 'qlvb_test';
const SIGNING_SECRET_KEY = process.env.SIGNING_SECRET_KEY || 'qlvb-test-signing-key-32chars-fixed-for-deterministic-tests';

// SAFETY: chặn reset DB không có 'test' trong tên
if (!PG_DATABASE.includes('test')) {
  console.error(`[SAFETY] PG_DATABASE='${PG_DATABASE}' khong chua 'test'. test-db-reset.ts chi reset DB co 'test' trong ten.`);
  process.exit(1);
}

const REPO_ROOT = path.resolve(__dirname, '../../..'); // backend/tools → backend → e_office_app_new → ROOT
const DB_DIR = path.join(REPO_ROOT, 'e_office_app_new', 'database');

  path.join(DB_DIR, 'init', '01_create_schemas.sql'),

  path.join(DB_DIR, 'seed', '001_required_data.sql'),
  path.join(DB_DIR, 'seed', '003_test_fixtures.sql'),
];

async function main() {
  const start = Date.now();
  console.log(`[test-db-reset] Target DB: ${PG_DATABASE}@${PG_HOST}:${PG_PORT}`);

  // Verify SQL files tồn tại
  for (const f of SQL_FILES) {
    if (!fs.existsSync(f)) {
      throw new Error(`SQL file not found: ${f}`);
    }
  }

  // 1. Connect to system DB (postgres) để DROP/CREATE qlvb_test
  const sysClient = new Client({
    host: PG_HOST, port: PG_PORT, user: PG_USER, password: PG_PASSWORD,
    database: 'postgres',
  });
  await sysClient.connect();

  try {
    console.log(`[test-db-reset] DROP DATABASE IF EXISTS ${PG_DATABASE} (force disconnect)...`);
    await sysClient.query(`SELECT pg_terminate_backend(pid) FROM pg_stat_activity WHERE datname = '${PG_DATABASE}' AND pid <> pg_backend_pid()`);
    await sysClient.query(`DROP DATABASE IF EXISTS ${PG_DATABASE}`);

    console.log(`[test-db-reset] CREATE DATABASE ${PG_DATABASE}...`);
    await sysClient.query(`CREATE DATABASE ${PG_DATABASE} WITH ENCODING='UTF8' LC_COLLATE='C' LC_CTYPE='C' TEMPLATE template0`);
  } finally {
    await sysClient.end();
  }

  // 2. Apply 4 SQL files via psql CLI (faster than pg lib for big schema files)
  const psqlBase = `psql -h ${PG_HOST} -p ${PG_PORT} -U ${PG_USER} -d ${PG_DATABASE} -v ON_ERROR_STOP=1 --set=PGPASSWORD=${PG_PASSWORD}`;
  // Set env vars cho psql
  const env = { ...process.env, PGPASSWORD: PG_PASSWORD };

  for (const sqlFile of SQL_FILES) {
    const fileName = path.basename(sqlFile);
    const t0 = Date.now();
    console.log(`[test-db-reset] Applying ${fileName}...`);

    // For seed 001 + 003, prepend SET commands
    let cmd: string;
    if (fileName.includes('seed/')) {
      cmd = `psql -h ${PG_HOST} -p ${PG_PORT} -U ${PG_USER} -d ${PG_DATABASE} -v ON_ERROR_STOP=1 -c "SET app.environment='test'; SET app.signing_secret_key='${SIGNING_SECRET_KEY}';" -f "${sqlFile}"`;
    } else if (fileName === '001_required_data.sql' || fileName === '003_test_fixtures.sql') {
      cmd = `psql -h ${PG_HOST} -p ${PG_PORT} -U ${PG_USER} -d ${PG_DATABASE} -v ON_ERROR_STOP=1 -c "SET app.environment='test'; SET app.signing_secret_key='${SIGNING_SECRET_KEY}';" -f "${sqlFile}"`;
    } else {
      cmd = `psql -h ${PG_HOST} -p ${PG_PORT} -U ${PG_USER} -d ${PG_DATABASE} -v ON_ERROR_STOP=1 -f "${sqlFile}"`;
    }

    try {
      execSync(cmd, { env, stdio: ['ignore', 'pipe', 'pipe'] });
      console.log(`[test-db-reset]   ${fileName} OK (${Date.now() - t0}ms)`);
    } catch (e) {
      const err = e as { stderr?: Buffer; stdout?: Buffer };
      console.error(`[test-db-reset]   FAILED: ${fileName}`);
      if (err.stderr) console.error(err.stderr.toString());
      if (err.stdout) console.error(err.stdout.toString());
      throw e;
    }
  }

  // 3. Verify fixtures
  const verifyClient = new Client({
    host: PG_HOST, port: PG_PORT, user: PG_USER, password: PG_PASSWORD,
    database: PG_DATABASE,
  });
  await verifyClient.connect();
  try {
    const r1 = await verifyClient.query(`SELECT COUNT(*) FROM public.staff WHERE id IN (9001, 9002, 9003, 9004, 9005, 9099)`);
    const r2 = await verifyClient.query(`SELECT COUNT(*) FROM edoc.incoming_docs WHERE id BETWEEN 90001 AND 90005`);
    const r3 = await verifyClient.query(`SELECT COUNT(*) FROM public.role_of_staff WHERE staff_id IN (9001, 9002, 9003, 9004, 9005, 9099)`);

    console.log(`[test-db-reset] VERIFY: staff fixtures = ${r1.rows[0].count} (expect 6)`);
    console.log(`[test-db-reset] VERIFY: incoming_docs fixtures = ${r2.rows[0].count} (expect 5)`);
    console.log(`[test-db-reset] VERIFY: role_of_staff fixtures = ${r3.rows[0].count} (expect 6)`);

    if (Number(r1.rows[0].count) < 6 || Number(r2.rows[0].count) < 5) {
      throw new Error('Verify failed — fixtures không đủ');
    }
  } finally {
    await verifyClient.end();
  }

  const elapsed = Date.now() - start;
  console.log(`[test-db-reset] DONE in ${elapsed}ms (target < 30000ms)`);
  if (elapsed > 30000) {
    console.warn(`[test-db-reset] WARNING: vượt 30s target (AUTO-02 acceptance)`);
  }
}

main().catch(err => {
  console.error('[test-db-reset] FATAL:', err.message || err);
  process.exit(1);
});

/**
 * tests/helpers/db.ts — DB helpers cho integration tests
 *
 * Pattern test isolation per AUTO-09:
 *   beforeEach(async () => { client = await getTestClient(); await client.query('BEGIN'); });
 *   afterEach(async () => { await client.query('ROLLBACK'); client.release(); });
 *
 * Hoặc dùng helper:
 *   await withTransaction(async (client) => {
 *     // test logic — auto rollback at end
 *   });
 */

import { Pool, PoolClient } from 'pg';

let testPool: Pool | null = null;

export function getTestPool(): Pool {
  if (!testPool) {
    testPool = new Pool({
      host: process.env.PG_HOST || 'localhost',
      port: Number(process.env.PG_PORT) || 5432,
      user: process.env.PG_USER || 'qlvb_admin',
      password: process.env.PG_PASSWORD || 'QlvbDev@2026',
      database: process.env.PG_DATABASE || 'qlvb_test',
      max: 10,
    });

    // Verify đang dùng test DB
    if (!testPool.options.database?.includes('test')) {
      throw new Error(`[TEST SAFETY] Test pool đang trỏ tới '${testPool.options.database}', không có 'test' trong tên`);
    }
  }
  return testPool;
}

export async function getTestClient(): Promise<PoolClient> {
  return await getTestPool().connect();
}

/**
 * Wrap callback trong transaction, auto ROLLBACK ở cuối (kể cả khi callback throw).
 * Dùng cho integration test cần verify DB state mà không persist data.
 */
export async function withTransaction<T>(
  callback: (client: PoolClient) => Promise<T>
): Promise<T> {
  const client = await getTestClient();
  try {
    await client.query('BEGIN');
    const result = await callback(client);
    return result;
  } finally {
    try {
      await client.query('ROLLBACK');
    } catch {
      // ignore rollback error (kết nối đã đóng)
    }
    client.release();
  }
}

/**
 * Đóng pool khi suite end. Gọi trong afterAll() cuối cùng.
 */
export async function closeTestPool(): Promise<void> {
  if (testPool) {
    await testPool.end();
    testPool = null;
  }
}
