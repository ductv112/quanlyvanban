```sql
-- ============================================================================
-- Phase 33: Seed LGSP placeholder data (REQ LGSP-CRED-05)
-- Source: docs/Trục EDOC Lạng Sơn - QLVB Doanh nghiệp/QLVBDNAgencies.xlsx + List.txt

-- ─── Phase 33.1: UPDATE 6 root unit set lgsp_org_code (theo mapping user approved 2026-XX-XX) ───
-- Mapping ID ↔ lgsp_org_code (đọc từ .planning/phases/33-database-core-infrastructure/33-02-root-units-mapping.md)
-- Idempotent: dùng UPDATE WHERE (id, lgsp_org_code IS DISTINCT FROM '...') để không overwrite nếu admin đã đổi
DO $$
DECLARE
  v_count_updated INT := 0;
BEGIN
  -- DN.001 — Cty CP Hữu nghị Xuân Cương
  UPDATE public.departments SET lgsp_org_code = 'H37.DN.001'
    WHERE id = {DEPT_ID_DN_001}  -- ← Replace with actual id from mapping (or NULL skip nếu chưa tồn tại)
      AND parent_id IS NULL
      AND (lgsp_org_code IS NULL OR lgsp_org_code = '');
  GET DIAGNOSTICS v_count_updated = ROW_COUNT;
  IF v_count_updated = 0 THEN RAISE NOTICE 'Phase 33 seed: DN.001 chưa được UPDATE (id không match root, hoặc đã có lgsp_org_code khác)'; END IF;

  -- DN.002 — Cty CP Sản xuất và Thương Mại Lạng Sơn
  UPDATE public.departments SET lgsp_org_code = 'H37.DN.002'
    WHERE id = {DEPT_ID_DN_002} AND parent_id IS NULL AND (lgsp_org_code IS NULL OR lgsp_org_code = '');
  GET DIAGNOSTICS v_count_updated = ROW_COUNT;
  IF v_count_updated = 0 THEN RAISE NOTICE 'Phase 33 seed: DN.002 chưa được UPDATE'; END IF;

  -- DN.003 — Cty CP Tập đoàn ĐT & XD Phú Lộc
  UPDATE public.departments SET lgsp_org_code = 'H37.DN.003'
    WHERE id = {DEPT_ID_DN_003} AND parent_id IS NULL AND (lgsp_org_code IS NULL OR lgsp_org_code = '');
  GET DIAGNOSTICS v_count_updated = ROW_COUNT;
  IF v_count_updated = 0 THEN RAISE NOTICE 'Phase 33 seed: DN.003 chưa được UPDATE'; END IF;

  -- DN.004 — Cty CP Kim loại màu Bắc Bộ
  UPDATE public.departments SET lgsp_org_code = 'H37.DN.004'
    WHERE id = {DEPT_ID_DN_004} AND parent_id IS NULL AND (lgsp_org_code IS NULL OR lgsp_org_code = '');
  GET DIAGNOSTICS v_count_updated = ROW_COUNT;
  IF v_count_updated = 0 THEN RAISE NOTICE 'Phase 33 seed: DN.004 chưa được UPDATE'; END IF;

  -- DN.005 — Cty TNHH TM XD Thiên Phú
  UPDATE public.departments SET lgsp_org_code = 'H37.DN.005'
    WHERE id = {DEPT_ID_DN_005} AND parent_id IS NULL AND (lgsp_org_code IS NULL OR lgsp_org_code = '');
  GET DIAGNOSTICS v_count_updated = ROW_COUNT;
  IF v_count_updated = 0 THEN RAISE NOTICE 'Phase 33 seed: DN.005 chưa được UPDATE'; END IF;

  -- DN.006 — Cty TNHH MTV Xe điện DK Việt Nhật
  UPDATE public.departments SET lgsp_org_code = 'H37.DN.006'
    WHERE id = {DEPT_ID_DN_006} AND parent_id IS NULL AND (lgsp_org_code IS NULL OR lgsp_org_code = '');
  GET DIAGNOSTICS v_count_updated = ROW_COUNT;
  IF v_count_updated = 0 THEN RAISE NOTICE 'Phase 33 seed: DN.006 chưa được UPDATE'; END IF;

  RAISE NOTICE 'Phase 33 seed: 6 root unit lgsp_org_code UPDATE done (mỗi UPDATE skip nếu id không match hoặc cột đã set)';
END $$;
```
