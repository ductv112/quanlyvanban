    ```tsx
    'use client';

    import React from 'react';
    import { Tag, Tooltip, Select } from 'antd';
    import { ApiOutlined, EditOutlined, TeamOutlined } from '@ant-design/icons';

    // ============================================================
    // LGSP Source Badge + Filter - shared helper for Văn bản đến (Phase 35)
    //
    // Used by:
    //   - frontend/src/app/(main)/van-ban-den/page.tsx (list rows + filter dropdown)
    //   - frontend/src/app/(main)/van-ban-den/[id]/page.tsx (detail Nguồn section)
    //
    // Source types from edoc.doc_source_type enum:
    //   'manual'         → Tạo thủ công bởi văn thư
    //   'internal'       → Sinh tự động từ "Gửi nội bộ" (Phase 17)
    //   'external_lgsp'  → Nhập từ trục LGSP (Phase 35 worker — REQ LGSP-RECV-05)
    //   'auto_from_sent' → Sinh tự động khi văn bản đi được ban hành (Phase 17)
    // ============================================================

    export type DocSourceType = 'manual' | 'internal' | 'external_lgsp' | 'auto_from_sent';

    interface LgspSourceBadgeProps {
      sourceType: DocSourceType | null | undefined;
      lgspSenderOrgCode?: string | null;
      publishUnit?: string | null;
      /** Khi true (mặc định): chỉ render khi sourceType là 'external_lgsp' (cho list column).
       *  Khi false: luôn render badge tương ứng cho mọi loại nguồn (cho detail section). */
      onlyShowLgsp?: boolean;
    }

    /**
     * Tag hiển thị nguồn của văn bản đến. Mặc định chỉ render khi sourceType='external_lgsp' (cho list).
     */
    export function LgspSourceBadge(props: LgspSourceBadgeProps): React.ReactElement | null {
      const { sourceType, lgspSenderOrgCode, publishUnit, onlyShowLgsp = true } = props;

      if (sourceType === 'external_lgsp') {
        const tooltipText = lgspSenderOrgCode
          ? `${publishUnit ?? 'Cơ quan ngoài'} (${lgspSenderOrgCode})`
          : (publishUnit ?? 'Văn bản từ trục LGSP');
        return (
          <Tooltip title={tooltipText}>
            <Tag color="blue" icon={<ApiOutlined />} style={{ marginInlineEnd: 0 }}>
              LGSP
            </Tag>
          </Tooltip>
        );
      }

      if (onlyShowLgsp) return null;

      if (sourceType === 'internal' || sourceType === 'auto_from_sent') {
        return (
          <Tag color="green" icon={<TeamOutlined />} style={{ marginInlineEnd: 0 }}>
            Nội bộ
          </Tag>
        );
      }

      if (sourceType === 'manual') {
        return (
          <Tag color="default" icon={<EditOutlined />} style={{ marginInlineEnd: 0 }}>
            Thủ công
          </Tag>
        );
      }

      return null;
    }

    interface LgspSourceFilterProps {
      value: DocSourceType | null;
      onChange: (next: DocSourceType | null) => void;
      style?: React.CSSProperties;
    }

    /**
     * Filter dropdown 4 lựa chọn (Tất cả / Nội bộ / LGSP / Thủ công).
     * 'Nội bộ' UI bao gồm cả 'internal' và 'auto_from_sent' — backend nhận 'internal' và filter group này.
     *
     * NOTE: hiện backend chỉ filter chính xác theo 1 source_type enum. Cho v3.2:
     *   - 'Nội bộ' UI → gửi source_type=internal (auto_from_sent thực tế ít — chấp nhận trade-off).
     *   - Nếu KH yêu cầu gộp 2 enum, Phase 37+ extend backend filter dùng IN clause.
     */
    export function LgspSourceFilter(props: LgspSourceFilterProps): React.ReactElement {
      const { value, onChange, style } = props;
      return (
        <Select
          allowClear
          placeholder="Tất cả nguồn"
          value={value ?? undefined}
          onChange={(next) => onChange((next as DocSourceType | undefined) ?? null)}
          style={{ minWidth: 160, ...style }}
          options={[
            { value: 'internal', label: 'Nội bộ' },
            { value: 'external_lgsp', label: 'LGSP' },
            { value: 'manual', label: 'Thủ công' },
          ]}
        />
      );
    }
    ```
