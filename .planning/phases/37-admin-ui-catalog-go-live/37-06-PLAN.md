    ```sql
    -- ============================================================
    -- Phase 37 Plan 37-06: Extend recipients SP với lgsp_tracking_id
    -- ============================================================
    DO $$
    BEGIN
      RAISE NOTICE 'Phase 37-06 schema: extend fn_outgoing_doc_get_recipients voi lgsp_tracking_id';
    END$$;

    DROP FUNCTION IF EXISTS edoc.fn_outgoing_doc_get_recipients(bigint) CASCADE;
    -- (Hoặc signature exact tìm được ở BƯỚC 1)

    CREATE OR REPLACE FUNCTION edoc.fn_outgoing_doc_get_recipients(
      p_outgoing_doc_id BIGINT
    ) RETURNS TABLE (
      id                          BIGINT,
      recipient_type              VARCHAR,
      recipient_unit_id           BIGINT,
      recipient_unit_name         VARCHAR,
      recipient_org_id            BIGINT,
      recipient_org_name          VARCHAR,
      recipient_org_code          VARCHAR,
      sent_at                     TIMESTAMPTZ,
      sent_status                 VARCHAR,
      error_message               TEXT,
      generated_incoming_doc_id   BIGINT,
      lgsp_doc_id                 VARCHAR,
      lgsp_status                 VARCHAR,
      lgsp_error_message          TEXT,
      lgsp_tracking_id            BIGINT       -- Phase 37 Plan 37-06: NEW column for retry
    ) AS $$
      SELECT r.id::BIGINT,
             r.recipient_type,
             r.recipient_unit_id::BIGINT,
             COALESCE(d.name, '')::VARCHAR,
             r.recipient_org_id::BIGINT,
             COALESCE(io.name, r.recipient_org_name, '')::VARCHAR,
             COALESCE(io.code, r.recipient_org_code, '')::VARCHAR,
             r.sent_at,
             r.sent_status::VARCHAR,
             r.error_message,
             r.generated_incoming_doc_id::BIGINT,
             t.lgsp_doc_id,
             t.status::VARCHAR              AS lgsp_status,
             t.error_message                AS lgsp_error_message,
             t.id::BIGINT                   AS lgsp_tracking_id
        FROM edoc.outgoing_doc_recipients r
        LEFT JOIN public.departments d         ON d.id = r.recipient_unit_id
        LEFT JOIN edoc.inter_organizations io  ON io.id = r.recipient_org_id
        LEFT JOIN edoc.lgsp_tracking t
                 ON t.outgoing_doc_id = r.outgoing_doc_id
                AND r.recipient_type = 'external_org'
                AND t.dest_org_code = COALESCE(io.code, r.recipient_org_code)
       WHERE r.outgoing_doc_id = p_outgoing_doc_id
       ORDER BY r.id;
    $$ LANGUAGE SQL STABLE;
    ```
