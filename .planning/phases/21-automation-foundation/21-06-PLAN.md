    - name: Install root deps

      working-directory: ${{ github.workspace }}
      run: |
        if [ -f package.json ]; then
          NODE_ENV=development npm install
        fi

    - name: Install backend deps
      shell: bash
      working-directory: ${{ github.workspace }}/e_office_app_new/backend
      run: NODE_ENV=development npm ci

    - name: Install mocks deps
      shell: bash
      working-directory: ${{ github.workspace }}/tools/mocks
      run: NODE_ENV=development npm install

    - name: Install test-report deps
      shell: bash
      working-directory: ${{ github.workspace }}/tools/test-report
      run: NODE_ENV=development npm install

    - name: Cache Playwright browsers
      if: inputs.install-playwright == 'true'
      uses: actions/cache@v4
      id: pw-cache
      with:
        path: ~/.cache/ms-playwright
        key: playwright-${{ runner.os }}-${{ hashFiles('package-lock.json') }}
        restore-keys: playwright-${{ runner.os }}-

    - name: Install Playwright browsers (chromium only)
      if: inputs.install-playwright == 'true' && steps.pw-cache.outputs.cache-hit != 'true'
      shell: bash
      run: npx playwright install chromium --with-deps

    - name: Install Playwright system deps (always — even on cache hit)
      if: inputs.install-playwright == 'true' && steps.pw-cache.outputs.cache-hit == 'true'

      run: npx playwright install-deps chromium

**2. Tạo `tools/mocks/start.ci.ts`** (CI-friendly mode — log to file, exit sau ready):

```typescript
/**
 * start.ci.ts — Boot 3 mocks cho CI environment
 *
 * Khác start.ts:
 *  - Output stdout đi đâu cũng OK (CI capture)
 *  - Tail process detached (exit 0 sau khi 3 mock UP)
 *  - PID file lưu vào /tmp/qlvb-mocks.pid để CI cleanup
 *
 * Usage trong CI:
 *   tsx tools/mocks/start.ci.ts &
 *   sleep 3
 *   # Test code chạy với mocks UP
 *   # Cleanup ở step "if: always":
 *   kill $(cat /tmp/qlvb-mocks.pid) || true
 */

import { spawn } from 'child_process';
import path from 'path';
import { fileURLToPath } from 'url';
import http from 'http';
import fs from 'fs';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

interface MockSpec {
  name: string;
  file: string;
  port: number;
}

const MOCKS: MockSpec[] = [
  { name: 'smartca', file: 'smartca-mock.ts', port: 8181 },
  { name: 'mysign',  file: 'mysign-mock.ts',  port: 8182 },
  { name: 'lgsp',    file: 'lgsp-mock.ts',    port: 8183 },
];

const PID_FILE = '/tmp/qlvb-mocks.pid';
const LOG_FILE = '/tmp/qlvb-mocks.log';

async function waitForHealth(port: number, timeoutMs = 15000): Promise<boolean> {
  const url = `http://localhost:${port}/health`;
  const start = Date.now();
  while (Date.now() - start < timeoutMs) {
    const ok = await new Promise<boolean>((resolve) => {
      http.get(url, (res) => { resolve(res.statusCode === 200); res.resume(); })
          .on('error', () => resolve(false));
    });
    if (ok) return true;
    await new Promise(r => setTimeout(r, 200));
  }
  return false;
}

async function main() {
  const logStream = fs.createWriteStream(LOG_FILE, { flags: 'a' });
  const pids: number[] = [];

  for (const spec of MOCKS) {
    const file = path.join(__dirname, spec.file);
    const child = spawn('npx', ['tsx', file], {
      stdio: ['ignore', 'pipe', 'pipe'],
      env: { ...process.env },
      detached: true,
    });
    child.stdout?.pipe(logStream);
    child.stderr?.pipe(logStream);
    child.unref();
    if (child.pid) pids.push(child.pid);
    console.log(`[start.ci] spawned ${spec.name} pid=${child.pid}`);
  }

  // Wait all 3 health
  const t0 = Date.now();
  const results = await Promise.all(MOCKS.map(s => waitForHealth(s.port)));
  if (!results.every(r => r)) {
    console.error(`[start.ci] FAIL: not all mocks UP. results=${JSON.stringify(results)}`);
    console.error(`[start.ci] Check log: ${LOG_FILE}`);
    process.exit(1);
  }

  // Save PIDs
  fs.writeFileSync(PID_FILE, pids.join('\n'));
  console.log(`[start.ci] All 3 mocks UP in ${Date.now() - t0}ms (pids: ${pids.join(', ')})`);
  console.log(`[start.ci] Log: ${LOG_FILE}`);
  console.log(`[start.ci] PIDs file: ${PID_FILE}`);
  process.exit(0);
}

main().catch((e) => {
  console.error('[start.ci] FATAL:', e);
  process.exit(1);
});
```
