import { chromium, FullConfig } from '@playwright/test';
import path from 'path';
import fs from 'fs';
import http from 'http';

// Import TEST_USERS từ backend (relative path từ root)
// KHÔNG dùng symlink — copy const trực tiếp để tránh ESM resolve issue
const TEST_USERS_FOR_AUTH = {
  admin:    { key: 'admin',    username: 'test_admin',    password: 'Test@123' },
  vanthu:   { key: 'vanthu',   username: 'test_vanthu',   password: 'Test@123' },
  lanhdao:  { key: 'lanhdao',  username: 'test_lanhdao',  password: 'Test@123' },
  canbo:    { key: 'canbo',    username: 'test_canbo',    password: 'Test@123' },
  canboX:   { key: 'canboX',   username: 'test_canbo_x',  password: 'Test@123' },
} as const;

const BACKEND_URL = process.env.PLAYWRIGHT_API_URL || 'http://localhost:4000';
const FRONTEND_URL = process.env.PLAYWRIGHT_BASE_URL || 'http://localhost:3000';
const STORAGE_STATE_DIR = path.resolve(__dirname, '.auth');

async function waitForBackend(timeoutMs = 30000): Promise<void> {
  const url = `${BACKEND_URL}/api/health`;
  const start = Date.now();
  while (Date.now() - start < timeoutMs) {
    const ok = await new Promise<boolean>((resolve) => {
      http.get(url, (res) => {
        resolve(res.statusCode === 200);
        res.resume();
      }).on('error', () => resolve(false));
    });
    if (ok) return;
    await new Promise(r => setTimeout(r, 500));
  }
  throw new Error(`[globalSetup] Backend không phản hồi tại ${url} sau ${timeoutMs}ms. Đảm bảo backend đang chạy: cd e_office_app_new/backend && npm run dev`);
}

async function loginAndSaveState(user: { key: string; username: string; password: string }): Promise<void> {
  const browser = await chromium.launch();
  const context = await browser.newContext();
  const page = await context.newPage();

  try {
    // Login qua API trực tiếp (KHÔNG navigate UI để tránh phụ thuộc frontend route)
    const loginResponse = await page.request.post(`${BACKEND_URL}/api/auth/login`, {
      data: { username: user.username, password: user.password },
      headers: { 'Content-Type': 'application/json' },
    });

    if (!loginResponse.ok()) {
      const body = await loginResponse.text();
      throw new Error(`Login fail cho ${user.username}: ${loginResponse.status()} ${body}`);
    }

    const json = await loginResponse.json();
    if (!json.success || !json.data?.accessToken) {
      throw new Error(`Login response invalid cho ${user.username}: ${JSON.stringify(json)}`);
    }

    const accessToken = json.data.accessToken;

    // Set localStorage cho frontend origin (axios interceptor sẽ đọc)
    await page.goto(FRONTEND_URL);
    await page.evaluate((token) => {
      window.localStorage.setItem('accessToken', token);
    }, accessToken);

    // Save storage state
    const statePath = path.join(STORAGE_STATE_DIR, `${user.key}.json`);
    await context.storageState({ path: statePath });
    console.log(`[globalSetup]   ${user.key.padEnd(10)} → ${statePath} OK`);
  } finally {
    await context.close();
    await browser.close();
  }
}

export default async function globalSetup(_config: FullConfig): Promise<void> {
  console.log('[globalSetup] Starting...');

  // Ensure auth dir exists
  if (!fs.existsSync(STORAGE_STATE_DIR)) {
    fs.mkdirSync(STORAGE_STATE_DIR, { recursive: true });
  }

  // 1. Wait for backend
  console.log(`[globalSetup] Waiting for backend at ${BACKEND_URL}/api/health...`);
  await waitForBackend();
  console.log('[globalSetup] Backend healthy ✓');

  // 2. Login 5 user parallel
  const users = Object.values(TEST_USERS_FOR_AUTH);
  const t0 = Date.now();
  await Promise.all(users.map(loginAndSaveState));
  console.log(`[globalSetup] Logged in ${users.length} users in ${Date.now() - t0}ms`);

  // 3. Verify 5 file generated
  for (const user of users) {
    const f = path.join(STORAGE_STATE_DIR, `${user.key}.json`);
    if (!fs.existsSync(f)) throw new Error(`Storage state file missing: ${f}`);
    const stat = fs.statSync(f);
    if (stat.size < 50) throw new Error(`Storage state file empty: ${f} (${stat.size} bytes)`);
  }
  console.log('[globalSetup] DONE');
}

export async function loginAs(
  request: { post: (url: string, opts: { data: unknown; headers?: Record<string, string> }) => Promise<{ ok(): boolean; json(): Promise<unknown> }> },
  role: RoleKey,
  baseUrl?: string
): Promise<{ accessToken: string; user: { id: number; username: string; full_name: string } }> {
  const apiUrl = baseUrl || process.env.PLAYWRIGHT_API_URL || 'http://localhost:4000';
  const credentials: Record<RoleKey, { username: string; password: string }> = {

    lanhdao:  { username: 'test_lanhdao',  password: 'Test@123' },
    canbo:    { username: 'test_canbo',    password: 'Test@123' },
    canboX:   { username: 'test_canbo_x',  password: 'Test@123' },
  };

  const cred = credentials[role];
  if (!cred) throw new Error(`Unknown role: ${role}`);

  const response = await request.post(`${apiUrl}/api/auth/login`, {
    data: cred,
    headers: { 'Content-Type': 'application/json' },
  });

    throw new Error(`Login response invalid: ${JSON.stringify(json)}`);

/**
 * tests/fixtures/api-client.ts — HTTP client wrapper cho integration tests
 *
 * Vitest integration test KHÔNG dùng supertest(app) trực tiếp vì backend server.ts
 * có app.listen() top-level. Workaround: gọi qua HTTP localhost:4000 (backend đã chạy bg).

 *

 *     let api: ApiClient;
 *     beforeAll(async () => {
 *       const { accessToken } = await loginAndGetToken('vanthu');
 *       api = createApiClient(accessToken);
 *     });
 *     it('TC-VBD-001 GET list', async () => {
 *       const res = await api.get('/api/van-ban-den');
 *       expect(res.status).toBe(200);
 *     });
 *   });
 */

const BACKEND_URL = process.env.PLAYWRIGHT_API_URL || process.env.BACKEND_URL || 'http://localhost:4000';

export interface ApiResponse<T = unknown> {
  status: number;
  ok: boolean;
  data: T;
  headers: Record<string, string>;
}

export interface ApiClient {
  get<T = unknown>(path: string): Promise<ApiResponse<T>>;
  post<T = unknown>(path: string, body?: unknown): Promise<ApiResponse<T>>;
  put<T = unknown>(path: string, body?: unknown): Promise<ApiResponse<T>>;
  patch<T = unknown>(path: string, body?: unknown): Promise<ApiResponse<T>>;
  delete<T = unknown>(path: string): Promise<ApiResponse<T>>;
  upload<T = unknown>(path: string, formData: FormData): Promise<ApiResponse<T>>;
}

export function createApiClient(accessToken?: string, baseUrl?: string): ApiClient {
  const base = baseUrl || BACKEND_URL;
  const baseHeaders: Record<string, string> = {};
  if (accessToken) baseHeaders['Authorization'] = `Bearer ${accessToken}`;

  async function request<T>(method: string, path: string, body?: unknown, isMultipart = false): Promise<ApiResponse<T>> {
    const url = path.startsWith('http') ? path : `${base}${path}`;
    const headers: Record<string, string> = { ...baseHeaders };
    let bodyToSend: BodyInit | undefined = undefined;

    if (body !== undefined) {
      if (isMultipart) {
        bodyToSend = body as FormData;
      } else {
        headers['Content-Type'] = 'application/json';
        bodyToSend = JSON.stringify(body);
      }
    }

    const res = await fetch(url, { method, headers, body: bodyToSend });
    const respHeaders: Record<string, string> = {};
    res.headers.forEach((v, k) => { respHeaders[k] = v; });

    let data: unknown;
    const contentType = res.headers.get('content-type') || '';
    if (contentType.includes('application/json')) {
      data = await res.json();
    } else {
      data = await res.text();
    }

    return {
      status: res.status,
      ok: res.ok,
      data: data as T,
      headers: respHeaders,
    };
  }

  return {
    get:    <T>(p: string) => request<T>('GET', p),
    post:   <T>(p: string, b?: unknown) => request<T>('POST', p, b),
    put:    <T>(p: string, b?: unknown) => request<T>('PUT', p, b),
    patch:  <T>(p: string, b?: unknown) => request<T>('PATCH', p, b),
    delete: <T>(p: string) => request<T>('DELETE', p),
    upload: <T>(p: string, fd: FormData) => request<T>('POST', p, fd, true),
  };
}

): Promise<{ accessToken: string; user: { id: number; username: string; full_name: string; roles?: string[] } }> {
  const credentials = {
    admin:    { username: 'test_admin',    password: 'Test@123' },
    vanthu:   { username: 'test_vanthu',   password: 'Test@123' },
    lanhdao:  { username: 'test_lanhdao',  password: 'Test@123' },
    canbo:    { username: 'test_canbo',    password: 'Test@123' },
    canboX:   { username: 'test_canbo_x',  password: 'Test@123' },
  };

  const cred = credentials[role];
  const apiClient = createApiClient();
  const res = await apiClient.post<{ success: boolean; data: { accessToken: string; user: { id: number; username: string; full_name: string; roles?: string[] } } }>(

  );

  if (!res.ok || !res.data.success) {
    throw new Error(`Login fail cho ${role} (${cred.username}): ${res.status}`);
  }

  return res.data.data;
}
